use-counter.ts
import { useState, useCallback } from "react";

export function useCounter(initial = 0) {
  const [count, setCount] = useState(initial);
  const increment = useCallback(() => {
    setCount(c => c + 1);
  }, []);
  return { count, increment, setCount };
}

Patterns you'll actually use.

Typing a component's props
tsx
type ButtonProps = {
  variant?: "primary" | "ghost"
  size?: "sm" | "md" | "lg"
  onClick?: (e: React.MouseEvent<HTMLButtonElement>) => void
  children: React.ReactNode
}

export function Button({
  variant = "primary",
  size = "md",
  onClick,
  children,
}: ButtonProps) {
  return (
    <button
      className={`btn ${variant} ${size}`}
      onClick={onClick}
    >
      {children}
    </button>
  )
}
A typed custom hook
ts
export function useLocalStorage<T>(
  key: string,
  initial: T
) {
  const [value, setValue] = useState<T>(() => {
    try {
      const raw = localStorage.getItem(key)
      return raw ? (JSON.parse(raw) as T) : initial
    } catch {
      return initial
    }
  })

  useEffect(() => {
    try {
      localStorage.setItem(key, JSON.stringify(value))
    } catch {}
  }, [key, value])

  return [value, setValue] as const
}
Discriminated union for state
ts
type State =
  | { status: "idle" }
  | { status: "loading" }
  | { status: "done"; data: User }
  | { status: "error"; message: string }

function render(s: State) {
  switch (s.status) {
    case "idle":    return <Placeholder />
    case "loading": return <Spinner />
    case "done":    return <Profile user={s.data} />
    case "error":   return <Error msg={s.message} />
  }
  // TS guarantees every case is handled.
}
